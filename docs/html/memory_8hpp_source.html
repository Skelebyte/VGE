83 bool internal_Malloc(size_t amount, const String &file, const String &func,

95 allocationSize = typeSize * amount;
96
97 if (Memory::Get().logMallocSizes) {

102

104 if (!data) {

112
113 for (int i = 0; i < amount; i++) {

135 void internal_Free(const String &file, const String &func, uint32 line) {
136 switch (state) {
137 case UNALLOCATED:
138 Logger::internal_Log("You need to call Malloc first! State: UNALLOCATED",
139 file, func, line);
140 return;
141 break;
142 case ALLOCATED:
143 break;
144 case FREED:
145 Logger::internal_Log("You need to call Malloc first! State: FREED", file,
146 func, line);
147 return;
148 break;
149 default:
150 Logger::internal_Log("You need to call Malloc first! State: UNKNOWN",
151 file, func, line);
152 return;
153 break;
154 }
155
156 if (!data) {
157 Logger::internal_Log("Data is not valid!", file, func, line);
158 return;
159 }
160
161 for (int i = 0; i < count; i++) {
162 data[i].~T();
163 }
164
165 if (Memory::Get().logFreeSizes) {
166 Logger::internal_Log("Freeing " + ToString(allocationSize) +
167 " bytes of memory.",
168 file, func, line);
169 }
170 std::free(data);
171 data = nullptr;
172
173 Memory::DecreaseUsedMemory(allocationSize);
174
175 state = MemoryState::FREED;
176
177 allocationSize = 0;
178 count = 0;
179 }
